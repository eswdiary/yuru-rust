49/// Matroska (MKV) and WebM demultiplexer.
50///
51/// `MkvReader` implements a demuxer for the Matroska and WebM formats.
52pub struct MkvReader<'s> {
53    /// Iterator over EBML element headers
54    iter: EbmlIterator<MediaSourceStream<'s>, MkvSchema>,
55    media_info: MediaInfo,
56    tracks: Vec<Track>,
57    track_states: HashMap<u32, TrackState>,
58    attachments: Vec<Attachment>,
59    chapters: Option<ChapterGroup>,
60    metadata: MetadataLog,
61    cues: Option<CuesElement>,
62    current_cluster: Option<ClusterState>,
63    frames: VecDeque<Frame>,
64}

66#[derive(Copy, Clone, Debug)]
67struct ClusterState {
68    /// The cluster timestamp in Segment ticks..
69    timestamp: Option<SegmentTicks>,
70    /// The start position in bytes of the cluster.
71    start: u64,
72}

74impl<'s> MkvReader<'s> {
75    pub fn try_new(mss: MediaSourceStream<'s>, opts: FormatOptions) -> Result<Self> {
76        // Get the total length of the stream, if possible.
77        let (is_seekable, total_len) = (mss.is_seekable(), mss.byte_len());
78
79        match total_len {
80            Some(len) if is_seekable => info!("stream is seekable with len={len} bytes."),
81            _ => (),
82        }
83
84        let mut it = EbmlIterator::new(mss, MkvSchema, total_len);
85
86        // Read the EBML header.
87        let ebml = it.next_element::<EbmlHeaderElement>()?;
88
89        if !matches!(ebml.doc_type.as_str(), "matroska" | "webm") {
90            return unsupported_error("mkv: not a matroska / webm file");
91        }
92
93        // Read the root element, Segment.
94        let segment_pos = match it.next_header()? {
95            Some(elem) if elem.element_type() == MkvElement::Segment => elem.data_pos(),
96            _ => return unsupported_error("mkv: missing segment element"),
97        };
98
99        // Descend into the Segment element.
100        it.push_element()?;
101
102        let mut segment_tracks = None;
103        let mut info = None;
104        let mut cues = None;
105        let mut current_cluster = None;
106        let mut seek_positions = Vec::new();
107        let mut tags = Vec::new();
108        let mut attachments = None;
109        let mut chapters = None;
110
111        while let Ok(Some(header)) = it.next_header() {
112            match header.element_type() {
113                MkvElement::SeekHead => {
114                    let seek_head = it.read_master_element::<SeekHeadElement>()?;
115                    for element in seek_head.seeks.into_vec() {
116                        let element_type = match it.schema().get_element_info(element.id as u32) {
117                            Some(info) => info.element_type(),
118                            None => continue,
119                        };
120                        seek_positions.push((element_type, element.position));
121                    }
122                }
123                MkvElement::Tracks => {
124                    segment_tracks = Some(it.read_master_element::<TracksElement>()?);
125                }
126                MkvElement::Info => {
127                    info = Some(it.read_master_element::<InfoElement>()?);
128                }
129                MkvElement::Cues => {
130                    cues = Some(it.read_master_element::<CuesElement>()?);
131                }
132                MkvElement::Tags => {
133                    // Multiple tags element per segment allowed.
134                    tags.push(it.read_master_element::<TagsElement>()?);
135                }
136                MkvElement::Cluster => {
137                    // Set state for current cluster for the first call of `next_element`.
138                    current_cluster =
139                        Some(ClusterState { timestamp: None, start: header.pos() - segment_pos });
140
141                    // Don't look forward into the stream since we can't be sure that we'll
142                    // find anything useful.
143                    break;
144                }
145                MkvElement::Attachments => {
146                    // Only one attachments element per segment is expected.
147                    if attachments.is_some() {
148                        log::warn!("unexpected attachments element");
149                    }
150                    attachments = Some(it.read_master_element::<AttachmentsElement>()?);
151                }
152                MkvElement::Chapters => {
153                    // Only one chapters element per segment is expected.
154                    if chapters.is_some() {
155                        log::warn!("unexpected chapters element");
156                    }
157                    chapters = Some(it.read_master_element::<ChaptersElement>()?);
158                }
159                other => {
160                    log::debug!("top-level scan ignored element {other:?}");
161                }
162            }
163        }
164
165        if is_seekable {
166            // All elements preceeding the element iterator's current position have already been
167            // read and do not need to be revisited.
168            seek_positions.retain(|sp| sp.1 >= it.pos());
169            // Make sure we don't jump backwards unnecessarily.
170            seek_positions.sort_by_key(|sp| sp.1);
171
172            for (_, pos) in seek_positions {
173                // Ascend back to the segment element.
174                it.pop_elements_upto(MkvElement::Segment)?;
175
176                // Seek the iterator to the child element.
177                it.seek_to_child(pos)?;
178
179                // Resume iteration.
180                let element_type = match it.next_header()? {
181                    Some(header) => header.element_type(),
182                    _ => continue,
183                };
184
185                // Safety: The element type or position may be incorrect. The element iterator will
186                // validate the type (as declared in the header) of the element at the seeked
187                // position against the element type asked to be read.
188                match element_type {
189                    MkvElement::Tracks => {
190                        segment_tracks = Some(it.read_master_element::<TracksElement>()?);
191                    }
192                    MkvElement::Info => {
193                        info = Some(it.read_master_element::<InfoElement>()?);
194                    }
195                    MkvElement::Tags => {
196                        // Multiple tags element per segment allowed.
197                        tags.push(it.read_master_element::<TagsElement>()?);
198                    }
199                    MkvElement::Cues => {
200                        cues = Some(it.read_master_element::<CuesElement>()?);
201                    }
202                    MkvElement::Attachments => {
203                        // Only one attachments element per segment is expected.
204                        if attachments.is_some() {
205                            log::warn!("unexpected attachments element after meta seek");
206                        }
207                        attachments = Some(it.read_master_element::<AttachmentsElement>()?);
208                    }
209                    MkvElement::Chapters => {
210                        // Only one chapters element per segment is expected.
211                        if chapters.is_some() {
212                            log::warn!("unexpected chapters element after meta seek");
213                        }
214                        chapters = Some(it.read_master_element::<ChaptersElement>()?)
215                    }
216                    _ => (),
217                }
218            }
219        }
220
221        let segment_tracks =
222            segment_tracks.ok_or(Error::DecodeError("mkv: missing Tracks element"))?;
223
224        // If seekable, seek to the start of the first cluster, if known, or the start of the
225        // segment. If unseekable, the element iterator is already positioned at the start of the
226        // first cluster.
227        if is_seekable {
228            let cluster_pos = current_cluster.as_ref().map(|cluster| cluster.start).unwrap_or(0);
229            it.seek_to_child(cluster_pos)?;
230            let _ = it.next_header()?;
231        }
232
233        // Descend into the cluster.
234        it.push_element()?;
235
236        let info = info.ok_or(Error::DecodeError("mkv: missing Info element"))?;
237
238        // Create a hashmap of all per-target tags (edition, chapter, & attachment tags).
239        let mut per_target_tags: TargetTagsMap = Default::default();
240
241        segment_tracks.get_target_uids(&mut per_target_tags);
242
243        if let Some(chapters) = &chapters {
244            chapters.get_target_uids(&mut per_target_tags);
245        }
246        if let Some(attachments) = &attachments {
247            attachments.get_target_uids(&mut per_target_tags);
248        }
249
250        // Begin with externally provided metadata and chapters.
251        let mut metadata = opts.external_data.metadata.unwrap_or_default();
252
253        // Post-process all tag elements into metadata revisions, while also collecting per-target
254        // tags.
255        let is_video = segment_tracks.tracks.as_ref().iter().any(|t| t.video.is_some());
256        for tag in tags {
257            metadata.push(tag.into_metadata(&mut per_target_tags, is_video));
258        }
259
260        // Post-process chapters element.
261        let chapters = chapters
262            .map(|chapters| chapters.into_chapter_group(&mut per_target_tags))
263            .unwrap_or(opts.external_data.chapters);
264
265        // Process attachments element.
266        let attachments = attachments
267            .map(|attachments| attachments.into_attachments(&mut per_target_tags))
268            .unwrap_or_default();
269
270        // Should TimeBase use a u64/u64 rational?
271        // Reduce the timebase to reduce the chance of overflows later.
272        let time_base = TimeBase::new(
273            info.timestamp_scale
274                .try_into()
275                .map_err(|_| Error::Unsupported("mkv: timestamp scale too large (report this)"))?,
276            NonZero::new(1_000_000_000).unwrap(),
277        )
278        .reduce();
279
280        let mut tracks = Vec::new();
281        let mut track_states = HashMap::new();
282
283        for track in segment_tracks.tracks {
284            // The track's timebase is scaled by the track timestamp scale.
285            let track_time_base = time_base
286                .scale(track.track_timestamp_scale)
287                .ok_or(Error::DecodeError("mkv: track timebase is invalid"))?;
288
289            // Create the track state.
290            let state = TrackState {
291                // TODO: This should be 64-bit, but track IDs are 32-bit.
292                track_num: u32::try_from(track.number.get())
293                    .map_err(|_| Error::Unsupported("mkv: track number too large (report this)"))?,
294                default_frame_duration: track.default_duration,
295                codec_delay: track.codec_delay,
296                track_time_base,
297                track_timestamp_scale: track.track_timestamp_scale,
298            };
299
300            // Create the track.
301            let mut tr = Track::new(state.track_num);
302
303            tr.with_time_base(track_time_base);
304
305            if let Some(lang_bcp47) = &track.lang_bcp47 {
306                tr.with_language(lang_bcp47);
307            }
308            else {
309                tr.with_language(&track.lang);
310            }
311
312            tr.with_flags(track.flags);
313
314            if let Some(codec_params) = make_track_codec_params(track)? {
315                tr.with_codec_params(codec_params);
316            }
317
318            tracks.push(tr);
319            track_states.insert(state.track_num, state);
320        }
321
322        // Populate media information.
323        let mut media_info = MediaInfo::new();
324
325        media_info.with_time_base(time_base);
326
327        if let Some(duration) = info.duration {
328            media_info.with_duration(Duration::new(duration.get().round() as u64));
329        }
330
331        Ok(Self {
332            iter: it,
333            media_info,
334            tracks,
335            track_states,
336            attachments,
337            chapters,
338            metadata,
339            cues,
340            current_cluster,
341            frames: VecDeque::new(),
342        })
343    }

345    fn seek_track_by_ts_forward(&mut self, track_id: u32, ts: Timestamp) -> Result<SeekedTo> {
346        let actual_ts = 'out: loop {
347            // Skip frames from the buffer until the given timestamp
348            while let Some(frame) = self.frames.front() {
349                let next_frame_pts = frame
350                    .pts
351                    .into_ts()
352                    .checked_add(frame.dur.into_dur())
353                    .ok_or(Error::SeekError(SeekErrorKind::OutOfRange))?;
354
355                if next_frame_pts >= ts && frame.track_num == track_id {
356                    break 'out frame.pts.into_ts();
357                }
358                else {
359                    self.frames.pop_front();
360                }
361            }
362
363            if !self.next_element()? {
364                // There are no more elements.
365                return Err(Error::SeekError(SeekErrorKind::OutOfRange));
366            }
367        };
368
369        Ok(SeekedTo { track_id, required_ts: ts, actual_ts })
370    }

372    fn seek_track_by_ts_atomic(
373        &mut self,
374        id: u32,
375        tb: TimeBase,
376        ts: Timestamp,
377    ) -> Result<SeekedTo> {
378        // Save the iterator and cluster states to restore in-case of and error.
379        let iter_state = self.iter.save_state();
380        let cluster_state = self.current_cluster;
381
382        match self.seek_track_by_ts(id, tb, ts) {
383            Err(err) => {
384                // Restore saved iterator and cluster states.
385                self.iter.restore_state(iter_state)?;
386                self.current_cluster = cluster_state;
387                Err(err)
388            }
389            value => value,
390        }
391    }
392
393    fn seek_track_by_ts(&mut self, id: u32, tb: TimeBase, ts: Timestamp) -> Result<SeekedTo> {
394        log::debug!("seeking track_id={id} to ts={ts}");
395
396        // If cues exist, seek to the nearest cue point.
397        if let Some(cues) = &self.cues {
398            let mut target_cue_point = None;
399
400            // Cue points store timestamps in Matroska ticks while the timestamp being seeked to is
401            // in signed Track ticks. Convert to unsigned Matroska ticks for iterating the cue
402            // points. If the timestamp is negative, then this is an error because cue points only
403            // contain unsigned Matroska ticks.
404            let ts = SignedTrackTicks::from(ts)
405                .try_into_matroska_ticks(tb)
406                .ok_or(Error::SeekError(SeekErrorKind::OutOfRange))?;
407
408            for cue_point in &cues.points {
409                if cue_point.time > ts {
410                    break;
411                }
412                target_cue_point = Some(cue_point);
413            }
414
415            let target_cue_point =
416                target_cue_point.ok_or(Error::SeekError(SeekErrorKind::OutOfRange))?;
417
418            log::debug!(
419                "found cue point: track_id={}, ts={}, seg_pos={}",
420                target_cue_point.positions.track.get(),
421                target_cue_point.time,
422                target_cue_point.positions.cluster_pos
423            );
424
425            // Ascend back to the segment element.
426            self.iter.pop_elements_upto(MkvElement::Segment)?;
427
428            // Seek to the specific cluster element.
429            self.iter.seek_to_child(target_cue_point.positions.cluster_pos)?;
430
431            // Resume iteration.
432            let cluster = match self.iter.next_header()? {
433                // The seeked element is a cluster.
434                Some(header) if header.element_type() == MkvElement::Cluster => header,
435                // The seeked element is not a cluster or there were no more elements at the cue
436                // position. The cue point was malformed.
437                _ => return seek_error(SeekErrorKind::Unseekable),
438            };
439
440            // Convert the cue point's timestamp (Matroska ticks) into Segment ticks for the cluster
441            // state.
442            // UNWRAP: The media information timestamp is always populated.
443            let timestamp =
444                target_cue_point.time.into_segment_ticks(self.media_info.time_base.unwrap());
445
446            // Update the current cluster metadata.
447            self.current_cluster =
448                Some(ClusterState { timestamp: Some(timestamp), start: cluster.pos() });
449
450            // Descend into the cluster element.
451            self.iter.push_element()?;
452
453            // If a cluster relative position is available, seek to the exact simple block or
454            // block group element.
455            if let Some(cluster_rel_pos) = target_cue_point.positions.cluster_rel_pos {
456                self.iter.seek_to_child(cluster_rel_pos)?;
457            }
458        }
459
460        // Seek to exact block.
461        self.seek_track_by_ts_forward(id, ts)
462    }
463
464    fn next_element(&mut self) -> Result<bool> {
465        match self.iter.next_header()? {
466            None => {
467                // The EBML iterator has consumed all child elements at the current level of the
468                // document.
469                match self.iter.parent() {
470                    None => {
471                        // The parent is the document. The media has ended.
472                        return Ok(false);
473                    }
474                    Some(parent) if parent.element_type() == MkvElement::Cluster => {
475                        // The parent was a cluster element. Reset the cluster state.
476                        self.current_cluster = None;
477                    }
478                    _ => (),
479                }
480
481                // Ascend to its parent.
482                self.iter.pop_element()?;
483            }
484            Some(child) => {
485                match child.element_type() {
486                    // Cluster element.
487                    MkvElement::Cluster => {
488                        self.current_cluster =
489                            Some(ClusterState { timestamp: None, start: child.data_pos() });
490
491                        // Descend into the cluster.
492                        self.iter.push_element()?;
493                    }
494                    // Children of a cluster element.
495                    MkvElement::Timestamp => {
496                        // Cluster timestamp element.
497                        match self.current_cluster.as_mut() {
498                            Some(cc) => {
499                                cc.timestamp = self.iter.read_u64()?.map(SegmentTicks::from)
500                            }
501                            _ => log::warn!("expected to have cluster"),
502                        }
503                    }
504                    block_type @ (MkvElement::SimpleBlock | MkvElement::BlockGroup) => {
505                        // Get the current cluster information.
506                        let Some(cluster) = self.current_cluster.as_ref()
507                        else {
508                            log::warn!("expected to have cluster");
509                            return Ok(true);
510                        };
511
512                        // Get the cluster timestamp.
513                        let Some(cluster_ts) = cluster.timestamp
514                        else {
515                            log::warn!("missing cluster timestamp");
516                            return Ok(true);
517                        };
518
519                        // Get block data and duration.
520                        let (data, duration) = match block_type {
521                            MkvElement::SimpleBlock => (self.iter.read_binary()?, None),
522                            MkvElement::BlockGroup => {
523                                let group = self.iter.read_master_element::<BlockGroupElement>()?;
524                                (group.data, group.duration)
525                            }
526                            _ => unreachable!(),
527                        };
528
529                        // Extract frames.
530                        if !extract_frames(
531                            &data,
532                            duration,
533                            cluster_ts,
534                            &self.track_states,
535                            &mut self.frames,
536                        )? {
537                            warn!("pts for block is too large");
538                            return Ok(false);
539                        }
540                    }
541                    // All other elements.
542                    other => {
543                        log::debug!("ignored element {other:?}");
544                    }
545                }
546            }
547        }
548
549        Ok(true)
550    }
551}

579    fn media_info(&self) -> &MediaInfo {
580        &self.media_info
581    }
582

584        &self.attachments
585    }
586
587    fn chapters(&self) -> Option<&ChapterGroup> {
588        self.chapters.as_ref()
589    }

591    fn metadata(&mut self) -> Metadata<'_> {
592        self.metadata.metadata()
593    }
594
595    fn seek(&mut self, _mode: SeekMode, to: SeekTo) -> Result<SeekedTo> {
596        if self.tracks.is_empty() {
597            return seek_error(SeekErrorKind::Unseekable);
598        }
599
600        match to {
601            SeekTo::Time { time, track_id } => {
602                let track = match track_id {
603                    Some(id) => self.tracks.iter().find(|track| track.id == id),
604                    None => self.tracks.first(),
605                };
606                let track = track.ok_or(Error::SeekError(SeekErrorKind::InvalidTrack))?;
607                // UNWRAP: Track will always have a timebase.
608                let tb = track.time_base.unwrap();
609                let ts = match tb.calc_timestamp(time) {
610                    Some(ts) => ts,
611                    None => {
612                        warn!("seek ts is too large");
613                        return seek_error(SeekErrorKind::OutOfRange);
614                    }
615                };
616                let track_id = track.id;
617                self.seek_track_by_ts_atomic(track_id, tb, ts)
618            }
619            SeekTo::Timestamp { ts, track_id } => {
620                match self.tracks.iter().find(|t| t.id == track_id) {
621                    Some(track) => {
622                        // UNWRAP: Track will always have a timebase.
623                        let tb = track.time_base.unwrap();
624                        self.seek_track_by_ts_atomic(track_id, tb, ts)
625                    }
626                    None => seek_error(SeekErrorKind::InvalidTrack),
627                }
628            }
629        }
630    }
631
632    fn tracks(&self) -> &[Track] {
633        &self.tracks
634    }
635
636    fn next_packet(&mut self) -> Result<Option<Packet>> {
637        loop {
638            if let Some(frame) = self.frames.pop_front() {
639                return Ok(Some(Packet::new(
640                    frame.track_num,
641                    frame.pts.into_ts(),
642                    frame.dur.into_dur(),
643                    frame.data,
644                )));
645            }
646
647            if !self.next_element()? {
648                // Reached the end of stream.
649                return Ok(None);
650            }
651        }
652    }
653
654    fn into_inner<'s>(self: Box<Self>) -> MediaSourceStream<'s>
655    where
656        Self: 's,
657    {
658        self.iter.into_inner()
659    }
660}

670impl From<EbmlError> for Error {
671    fn from(value: EbmlError) -> Self {
672        // All non-IO EBML errors are mapped to a decode error.
673        let msg = match value {
