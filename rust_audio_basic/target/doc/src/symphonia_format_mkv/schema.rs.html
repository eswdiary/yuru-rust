293/// The element is global.
294const EBML_GLOBAL: u8 = 1 << 7;
295/// The element may be nested in itself.
296const EBML_RECURSIVE: u8 = 1 << 6;
297/// The element may be an unknown size.
298const EBML_UNKNOWN: u8 = 1 << 5;
299/// Mask of flag bits storing element min. depth.
300const EBML_DEPTH_MASK: u8 = 0xf;
301
302/// MKV element information.
303///
304/// 4-tuple consisting of element type, element data type, flag bits, & element ID. 8 bytes/element.
305#[derive(Copy, Clone)]
306pub struct MkvElementInfo(MkvElement, EbmlDataType, u8, u32);
307
308impl EbmlElementInfo for MkvElementInfo {
309    type ElementType = MkvElement;
310
311    fn element_type(&self) -> Self::ElementType {
312        self.0
313    }
314
315    fn data_type(&self) -> EbmlDataType {
316        self.1
317    }
318
319    fn min_depth(&self) -> u8 {
320        self.2 & EBML_DEPTH_MASK
321    }
322
323    fn is_global(&self) -> bool {
324        self.2 & EBML_GLOBAL == EBML_GLOBAL
325    }
326
327    fn is_recursive(&self) -> bool {
328        self.2 & EBML_RECURSIVE == EBML_RECURSIVE
329    }
330
331    fn allow_unknown_size(&self) -> bool {
332        self.2 & EBML_UNKNOWN == EBML_UNKNOWN
333    }
334
335    fn parent_id(&self) -> u32 {
336        self.3
337    }
338}
