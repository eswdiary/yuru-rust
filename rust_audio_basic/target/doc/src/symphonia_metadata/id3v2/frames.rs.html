30// The following is a list of all standardized ID3v2.x frames for all ID3v2 major versions and their
31// implementation status ("S" column) in Symphonia.
32//
33// ID3v2.2 uses 3 character frame identifiers as opposed to the 4 character identifiers used in
34// subsequent versions. This table may be used to map equivalent frames between the two versions.
35//
36// All ID3v2.3 frames are officially part of ID3v2.4 with the exception of those marked "n/a".
37// However, it is likely that ID3v2.3-only frames appear in some real-world ID3v2.4 tags.

159/// An ID3v2 chapter.
160#[derive(Clone, Debug)]
161pub struct Id3v2Chapter {
162    // The chapter identifier.
163    pub id: String,
164    /// A counter indicating the order the chapter frame was read.
165    pub read_order: usize,
166    /// The chapter contents.
167    pub chapter: Chapter,
168}
169
170/// An ID3v2 table of contents describes different sections and chapters of an audio stream.
171#[derive(Clone, Debug, Default)]
172pub struct Id3v2TableOfContents {
173    /// The table of contents identifier.
174    pub id: String,
175    /// Indicates if this is the top-level table of contents frame. Only one table of contents
176    /// frame should be marked top-level, and not be a child of any other frame.
177    pub top_level: bool,
178    /// Indicates if the entries should be played as a continuous ordered sequence or played
179    /// individually.
180    ///
181    /// TODO: It is not clear if this is useful.
182    #[allow(dead_code)]
183    pub ordered: bool,
184    /// The identifiers of the items that belong to this table of contents. These may identify
185    /// a chapter or another table of contents.
186    pub items: Vec<String>,
187    /// The tags associated with this table of contents.
188    pub tags: Vec<Tag>,
189    /// The visuals associated with this table of contents.
190    pub visuals: Vec<Visual>,
191}
192
193/// The result of parsing a frame.
194pub enum FrameResult {
195    /// The frame was skipped.
196    Skipped,
197    /// Padding was encountered instead of a frame. The remainder of the ID3v2 tag may be skipped.
198    Padding,
199    /// A frame was parsed and yielded a single `Tag`.
200    Tag(Tag),
201    /// A frame was parsed and yielded a single `Visual`.
202    Visual(Visual),
203    /// A frame was parsed and yielded many `Tag`s.
204    MultipleTags(SmallVec<[Tag; 2]>),
205    /// A frame was parsed and yielded a chapter.
206    Chapter(Id3v2Chapter),
207    /// A frame was parsed and yielded a table of contents.
208    TableOfContents(Id3v2TableOfContents),
209}

211/// Gets the minimum frame size for a major version of an ID3v2.
212pub fn min_frame_size(major_version: u8) -> u64 {
213    match major_version {
214        2 => 6,
215        3 | 4 => 10,
216        _ => unreachable!("id2v3: unexpected version"),
217    }
218}

413/// Validates that a frame id only contains uppercase letters (A-Z), and digits (0-9).
414fn validate_frame_id(id: &[u8]) -> bool {
415    // Only frame IDs with 3 or 4 characters are valid.
416    if id.len() != 4 && id.len() != 3 {
417        return false;
418    }
419
420    id.iter().all(|b| b.is_ascii_uppercase() || b.is_ascii_digit())
421}

423/// Gets a slice of ASCII bytes as a string slice.
424///
425/// Assumes the bytes are valid ASCII characters. Panics otherwise.
426fn from_ascii(id: &[u8]) -> &str {
427    std::str::from_utf8(id).expect("ascii only")
428}

430/// Get the default frame reader for unknown frames.
431fn null_frame_reader() -> (FrameReader, Option<RawTagParser>) {
432    (read_raw_frame, None)
433}
434
435/// Find a frame reader and optional raw tag parser for modern ID3v2.3+ frames.
436fn find_frame_reader(id: [u8; 4]) -> (FrameReader, Option<RawTagParser>) {
437    FRAME_READERS.get(&id).map(|a| a.to_owned()).unwrap_or_else(|| null_frame_reader())
438}

440/// Find a frame reader and optional raw tag parser for legacy ID3v2.2 frames by finding an
441/// equivalent modern ID3v2.3+ frame reader.
442fn find_legacy_frame_reader(id: [u8; 3]) -> (FrameReader, Option<RawTagParser>) {
443    match LEGACY_FRAME_MAP.get(&id) {
444        Some(id) => find_frame_reader(**id),
445        _ => null_frame_reader(),
446    }
447}

449/// Read an ID3v2.2 frame.
450pub fn read_id3v2p2_frame<B: ReadBytes>(reader: &mut B) -> Result<FrameResult> {
451    let id = reader.read_triple_bytes()?;
452
453    // Check if the frame id contains valid characters. If it does not, then assume the rest of the
454    // tag is padding. As per the specification, padding should be all 0s, but there are some tags
455    // which don't obey the specification.
456    if !validate_frame_id(&id) {
457        // As per the specification, padding should be all 0s, but there are some tags which don't
458        // obey the specification.
459        if id != [0, 0, 0] {
460            warn!("padding bytes not zero");
461        }
462
463        return Ok(FrameResult::Padding);
464    }
465
466    let size = u64::from(reader.read_be_u24()?);
467
468    // Find a reader for the frame.
469    let (read_frame, raw_tag_parser) = find_legacy_frame_reader(id);
470
471    // A frame must be atleast 1 byte as per the specification.
472    if size == 0 {
473        warn!("'{}' was skipped because it has a size of 0", from_ascii(&id));
474        return Ok(FrameResult::Skipped);
475    }
476
477    // Read the frame body into a frame buffer.
478    let data = reader.read_boxed_slice_exact(size as usize)?;
479
480    // Prepare frame information for the frame reader.
481    let info = FrameInfo::new(&id, 2, raw_tag_parser);
482
483    // An error while reading the frame from the frame buffer is not fatal.
484    match read_frame(BufReader::new(&data), &info) {
485        Ok(result) => Ok(result),
486        Err(err) => {
487            // On error, skip the frame.
488            warn!("{err}");
489            Ok(FrameResult::Skipped)
490        }
491    }
492}

494/// Read an ID3v2.3 frame.
495pub fn read_id3v2p3_frame<B: ReadBytes>(reader: &mut B) -> Result<FrameResult> {
496    let id = reader.read_quad_bytes()?;
497
498    // Check if the frame id contains valid characters. If it does not, then assume the rest of the
499    // tag is padding. As per the specification, padding should be all 0s, but there are some tags
500    // which don't obey the specification.
501    if !validate_frame_id(&id) {
502        // As per the specification, padding should be all 0s, but there are some tags which don't
503        // obey the specification.
504        if id != [0, 0, 0, 0] {
505            warn!("padding bytes not zero");
506        }
507
508        return Ok(FrameResult::Padding);
509    }
510
511    // The size of the frame after encryption, compression, and unsynchronisation.
512    let size = reader.read_be_u32()?;
513
514    // Frame-specific flags.
515    let flags = reader.read_be_u16()?;
516
517    // Unused flag bits must be cleared.
518    if flags & 0x1f1f != 0x0 {
519        return decode_error("id3v2: unused flag bits are not cleared");
520    }
521
522    // Frame-specific flags that are important for reading.
523    let is_compressed = flags & 0x80 != 0;
524    let is_encrypted = flags & 0x40 != 0;
525    let is_grouped = flags & 0x20 != 0;
526
527    // When some flags are set, the frame header is extended with additional fields. Calculate the
528    // size of these fields.
529    let flag_data_size = if is_compressed { 4 } else { 0 } // 4-byte decompressed size.
530        + if is_encrypted { 1 } else { 0 } // 1-byte encryption ID.
531        + if is_grouped { 1 } else { 0 }; // 1-byte group ID.
532
533    // If the frame size is too small for the extended header, there is a fatal framing error.
534    if size < flag_data_size {
535        return decode_error("id3v2: the frame is too small");
536    }
537
538    // The size of the frame's body.
539    let data_size = size - flag_data_size;
540
541    // If compression is enabled, read the decompressed size of the frame.
542    let _decompressed_size = if is_compressed { Some(reader.read_be_u32()?) } else { None };
543
544    // If encryption is enabled, read the encryption ID. A sub-field indicating the frame is
545    // encrypted, and its encryption ID will be added to the tag.
546    let encryption_id = if is_encrypted { Some(reader.read_byte()?) } else { None };
547
548    // If frame grouping is enabled, read the group ID of the frame. A sub-field indicating the
549    // group will be added to the tag.
550    let group_id = if is_grouped { Some(reader.read_byte()?) } else { None };
551
552    // TODO: Implement zlib DEFLATE decompression.
553    if is_compressed {
554        reader.ignore_bytes(u64::from(data_size))?;
555
556        warn!("'{}' was skipped because compressed frames are not supported", from_ascii(&id));
557        return Ok(FrameResult::Skipped);
558    }
559
560    // A zero-length frame body is not allowed, but can be skipped.
561    if data_size == 0 {
562        warn!("'{}' was skipped because it has a size of 0", from_ascii(&id));
563        return Ok(FrameResult::Skipped);
564    }
565
566    // Read the frame body into a frame buffer.
567    let data = reader.read_boxed_slice_exact(data_size as usize)?;
568
569    // Find a reader for the frame. If the frame is encrypted, then the frame is provided as a
570    // binary buffer.
571    let (read_frame, raw_tag_parser) =
572        if !is_encrypted { find_frame_reader(id) } else { null_frame_reader() };
573
574    // Prepare frame information for the frame reader.
575    let info = FrameInfo::new(&id, 3, raw_tag_parser);
576
577    // Read the frame from the frame buffer. An error here is not fatal.
578    match read_frame(BufReader::new(&data), &info) {
579        Ok(mut result) => {
580            // Add the sub-fields to the tag if the frame is grouped or encrypted.
581            if is_grouped || is_encrypted {
582                append_tag_sub_fields(&mut result, group_id, encryption_id);
583            }
584            Ok(result)
585        }
586        Err(err) => {
587            // On error, skip the frame.
588            warn!("{err}");
589            Ok(FrameResult::Skipped)
590        }
591    }
592}

594/// Read an ID3v2.4 frame.
595pub fn read_id3v2p4_frame<B: ReadBytes + FiniteStream>(reader: &mut B) -> Result<FrameResult> {
596    let id = reader.read_quad_bytes()?;
597
598    // Check if the frame id contains valid characters. If it does not, then assume the rest of the
599    // tag is padding.
600    if !validate_frame_id(&id) {
601        // As per the specification, padding should be all 0s, but there are some tags which don't
602        // obey the specification.
603        if id != [0, 0, 0, 0] {
604            warn!("padding bytes not zero");
605        }
606
607        return Ok(FrameResult::Padding);
608    }
609
610    // The size of the frame after encryption, compression, and unsynchronisation.
611    let size = read_syncsafe_leq32(reader, 28)?;
612
613    // Frame-specific flags.
614    let flags = reader.read_be_u16()?;
615
616    // Unused flag bits must be cleared.
617    if flags & 0x8fb0 != 0x0 {
618        return decode_error("id3v2: unused flag bits are not cleared");
619    }
620
621    // Frame-specific flags that are important for reading.
622    let is_grouped = flags & 0x40 != 0;
623    let is_compressed = flags & 0x08 != 0;
624    let is_encrypted = flags & 0x04 != 0;
625    let is_unsynchronised = flags & 0x2 != 0;
626    let has_indicated_size = flags & 0x1 != 0;
627
628    if is_compressed && !has_indicated_size {
629        return decode_error("id3v2: frame compressed without a data length indicator");
630    }
631
632    // When some flags are set, the frame header is extended with additional fields. Calculate the
633    // size of these fields.
634    let flag_data_size = if is_grouped { 1 } else { 0 } // 1-byte group ID.
635        + if is_encrypted { 1 } else { 0 } // 1-byte encryption ID.
636        + if has_indicated_size { 4 } else { 0 }; // 4-byte data length indicator.
637
638    // If the frame size is too small for the extended header, there is a fatal framing error.
639    if size < flag_data_size {
640        return decode_error("id3v2: the frame is too small");
641    }
642
643    // The size of the frame's body.
644    let data_size = size - flag_data_size;
645
646    // Frame group identifier byte. Used to group a set of frames. The frame group will be added as
647    // a sub-field to all produced tags.
648    let group_id = if is_grouped { Some(reader.read_byte()?) } else { None };
649
650    // Frame encryption flag. Encryption is vendor-specific. Therefore, an encrypted frame will only
651    // be provided as a binary buffer. A sub-field indicating the frame is encrypted will be added.
652    let encryption_id = if is_encrypted { Some(reader.read_byte()?) } else { None };
653
654    // The data length indicator is optional in the frame header. This field indicates the original
655    // size of the frame body before compression, encryption, and/or unsynchronisation. It is
656    // mandatory if compression is used, but only encouraged for unsynchronisation, and optional for
657    // encryption.
658    //
659    // The indicated size will be added to all produced tags as a sub-field if the frame is
660    // encrypted.
661    let _indicated_size =
662        if has_indicated_size { Some(read_syncsafe_leq32(reader, 28)?) } else { None };
663
664    // TODO: Implement zlib DEFLATE decompression.
665    if is_compressed {
666        reader.ignore_bytes(u64::from(data_size))?;
667
668        warn!("'{}' was skipped because compressed frames are not supported", from_ascii(&id));
669        return Ok(FrameResult::Skipped);
670    }
671
672    // A zero-length frame body is not allowed, but can be skipped.
673    if data_size == 0 {
674        warn!("'{}' was skipped because it has a size of 0", from_ascii(&id));
675        return Ok(FrameResult::Skipped);
676    }
677
678    // Read the frame body into a frame buffer.
679    let mut data = reader.read_boxed_slice_exact(data_size as usize)?;
680
681    // Find a reader for the frame. If the frame is encrypted, then the frame is provided as a
682    // binary buffer.
683    let (read_frame, raw_tag_parser) =
684        if !is_encrypted { find_frame_reader(id) } else { null_frame_reader() };
685
686    // Prepare frame information for the frame reader.
687    let info = FrameInfo::new(&id, 4, raw_tag_parser);
688
689    // Read the frame.
690    let result = if is_unsynchronised {
691        // The frame body has been unsynchronised. Decode the unsynchronised data back to it's
692        // unencoded form in-place before parsing.
693        let unsync_data = decode_unsynchronisation(&mut data);
694
695        read_frame(BufReader::new(unsync_data), &info)
696    }
697    else {
698        // The frame body has not been unsynchronised.
699        read_frame(BufReader::new(&data), &info)
700    };
701
702    // An error while reading the frame from the frame buffer is not fatal.
703    match result {
704        Ok(mut result) => {
705            // Add the sub-fields to the tag if the frame is grouped or encrypted.
706            if is_grouped || is_encrypted {
707                append_tag_sub_fields(&mut result, group_id, encryption_id);
708            }
709            Ok(result)
710        }
711        Err(err) => {
712            // On error, skip the frame.
713            warn!("{err}");
714            Ok(FrameResult::Skipped)
715        }
716    }
717}
718
719fn append_tag_sub_fields(result: &mut FrameResult, grp_id: Option<u8>, enc_id: Option<u8>) {
720    let add_sub_fields = |tag: &mut Tag| {
721        // Take the existing boxed sub-fields, and turn it into a vector. Or, create a new
722        // vector.
723        let mut sub_fields = tag.raw.sub_fields.take().unwrap_or_default().into_vec();
724
725        if let Some(id) = grp_id {
726            sub_fields.push(RawTagSubField::new(GROUP_ID, RawValue::from(id)));
727        }
728
729        if let Some(id) = enc_id {
730            sub_fields.push(RawTagSubField::new(ENCRYPTION_METHOD_ID, RawValue::from(id)));
731        }
732
733        // Put back the extended sub-fields.
734        tag.raw.sub_fields.replace(sub_fields.into_boxed_slice());
735    };
736
737    match result {
738        FrameResult::Tag(tag) => add_sub_fields(tag),
739        FrameResult::MultipleTags(tags) => tags.iter_mut().for_each(add_sub_fields),
740        _ => (),
741    }
