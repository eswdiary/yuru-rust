39fn is_supported_adpcm_codec(codec_id: AudioCodecId) -> bool {
40    matches!(codec_id, CODEC_ID_ADPCM_MS | CODEC_ID_ADPCM_IMA_WAV | CODEC_ID_ADPCM_IMA_QT)
41}
42
43enum InnerDecoder {
44    AdpcmMs,
45    AdpcmIma,
46    AdpcmImaQT,
47}
48
49impl InnerDecoder {
50    fn decode_mono_fn<B: ReadBytes>(&self) -> impl Fn(&mut B, &mut [i32], usize) -> Result<()> {
51        match *self {
52            InnerDecoder::AdpcmMs => codec_ms::decode_mono,
53            InnerDecoder::AdpcmIma => codec_ima_wav::decode_mono,
54            InnerDecoder::AdpcmImaQT => codec_ima_qt::decode_mono,
55        }
56    }
57
58    fn decode_stereo_fn<B: ReadBytes>(
59        &self,
60    ) -> impl Fn(&mut B, [&mut [i32]; 2], usize) -> Result<()> {
61        match *self {
62            InnerDecoder::AdpcmMs => codec_ms::decode_stereo,
63            InnerDecoder::AdpcmIma => codec_ima_wav::decode_stereo,
64            InnerDecoder::AdpcmImaQT => codec_ima_qt::decode_stereo,
65        }
66    }
67}

69/// Adaptive Differential Pulse Code Modulation (ADPCM) decoder.
70pub struct AdpcmDecoder {
71    params: AudioCodecParameters,
72    inner_decoder: InnerDecoder,
73    buf: AudioBuffer<i32>,
74}
75
76impl AdpcmDecoder {
77    pub fn try_new(params: &AudioCodecParameters, _opts: &AudioDecoderOptions) -> Result<Self> {
78        // This decoder only supports certain ADPCM codecs.
79        if !is_supported_adpcm_codec(params.codec) {
80            return unsupported_error("adpcm: invalid codec");
81        }
82
83        let frames = match params.max_frames_per_packet {
84            Some(frames) => frames as usize,
85            _ => return unsupported_error("adpcm: maximum frames per packet is required"),
86        };
87
88        if params.frames_per_block.is_none() || params.frames_per_block.unwrap() == 0 {
89            return unsupported_error("adpcm: valid frames per block is required");
90        }
91
92        let rate = match params.sample_rate {
93            Some(rate) => rate,
94            _ => return unsupported_error("adpcm: sample rate is required"),
95        };
96
97        let spec = if let Some(channels) = &params.channels {
98            if channels.count() > 2 {
99                return unsupported_error("adpcm: up to two channels are supported");
100            }
101
102            AudioSpec::new(rate, channels.clone())
103        }
104        else {
105            return unsupported_error("adpcm: channels or channel_layout is required");
106        };
107
108        let inner_decoder = match params.codec {
109            CODEC_ID_ADPCM_MS => InnerDecoder::AdpcmMs,
110            CODEC_ID_ADPCM_IMA_WAV => InnerDecoder::AdpcmIma,
111            CODEC_ID_ADPCM_IMA_QT => InnerDecoder::AdpcmImaQT,
112            _ => return unsupported_error("adpcm: codec is unsupported"),
113        };
114
115        Ok(AdpcmDecoder {
116            params: params.clone(),
117            inner_decoder,
118            buf: AudioBuffer::new(spec, frames),
119        })
120    }
121
122    fn decode_inner(&mut self, packet: &PacketRef<'_>) -> Result<()> {
123        let mut stream = packet.as_buf_reader();
124
125        let frames_per_block = self.params.frames_per_block.unwrap() as usize;
126
127        let block_count = packet.block_dur().get() as usize / frames_per_block;
128
129        self.buf.clear();
130        self.buf.render_uninit(Some(block_count * frames_per_block));
131
132        let channel_count = self.buf.spec().channels().count();
133        match channel_count {
134            1 => {
135                let buffer = self.buf.plane_mut(0).unwrap();
136                let decode_mono = self.inner_decoder.decode_mono_fn();
137                for block_id in 0..block_count {
138                    let offset = frames_per_block * block_id;
139                    let buffer_range = offset..(offset + frames_per_block);
140                    let buffer = &mut buffer[buffer_range];
141                    decode_mono(&mut stream, buffer, frames_per_block)?;
142                }
143            }
144            2 => {
145                let buffers = self.buf.plane_pair_mut(0, 1).unwrap();
146                let decode_stereo = self.inner_decoder.decode_stereo_fn();
147                for block_id in 0..block_count {
148                    let offset = frames_per_block * block_id;
149                    let buffer_range = offset..(offset + frames_per_block);
150                    let buffers =
151                        [&mut buffers.0[buffer_range.clone()], &mut buffers.1[buffer_range]];
152                    decode_stereo(&mut stream, buffers, frames_per_block)?;
153                }
154            }
155            _ => unreachable!(),
156        }
157
158        Ok(())
159    }
160}
161
162impl AudioDecoder for AdpcmDecoder {
163    fn reset(&mut self) {
164        // No state is stored between packets, therefore do nothing.
165    }
166
167    fn codec_info(&self) -> &CodecInfo {
168        // Return the codec that's in-use.
169        &Self::supported_codecs().iter().find(|desc| desc.id == self.params.codec).unwrap().info
170    }
171
172    fn codec_params(&self) -> &AudioCodecParameters {
173        &self.params
174    }
175
176    fn decode_ref(&mut self, packet: &PacketRef<'_>) -> Result<GenericAudioBufferRef<'_>> {
177        if let Err(e) = self.decode_inner(packet) {
178            self.buf.clear();
179            Err(e)
180        }
181        else {
182            Ok(self.buf.as_generic_audio_buffer_ref())
183        }
184    }
185
186    fn finalize(&mut self) -> FinalizeResult {
187        Default::default()
188    }
189
190    fn last_decoded(&self) -> GenericAudioBufferRef<'_> {
191        self.buf.as_generic_audio_buffer_ref()
192    }
193}

195impl RegisterableAudioDecoder for AdpcmDecoder {
196    fn try_registry_new(
197        params: &AudioCodecParameters,
198        opts: &AudioDecoderOptions,
199    ) -> Result<Box<dyn AudioDecoder>>
200    where
201        Self: Sized,
202    {
203        Ok(Box::new(AdpcmDecoder::try_new(params, opts)?))
204    }
205
206    fn supported_codecs() -> &'static [SupportedAudioCodec] {
207        &[
208            support_audio_codec!(CODEC_ID_ADPCM_MS, "adpcm_ms", "Microsoft ADPCM"),
209            support_audio_codec!(CODEC_ID_ADPCM_IMA_WAV, "adpcm_ima_wav", "ADPCM IMA WAV"),
210            support_audio_codec!(CODEC_ID_ADPCM_IMA_QT, "adpcm_ima_qt", "ADPCM IMA QT"),
211        ]
212    }
